# 09. 도구 레시피 (docx, 엑셀을 직접 고칠 때 쓰는 짧은 코드)

전용 도구가 없어도 쓸 수 있게, 신청서 docx를 **사용자 수정과 서식을 지키면서** 고칠 때 필요한 처리를 짧은 파이썬으로 모았다. 필요한 것만 복사해 쓴다.

준비: `pip install python-docx lxml openpyxl pymupdf pillow`

원칙
- 원본 docx는 먼저 이력 폴더에 복사하고, 코드는 복사본을 읽어 **새 이름으로** 저장한다.
- 문단을 통째로 다시 만들지 말고, 바꿀 글자만 바꾼다(아래 1번). 서식, 메모 표시, 사용자가 칠한 색이 그대로 남는다.
- 저장 뒤 반드시 5번(검증)을 돌린다.

## 1. 문단 안 글자만 바꾸기 (런 경계를 넘어도 서식 유지)
Word는 한 문장을 여러 조각(런)으로 나눠 저장하므로 단순 치환이 안 될 때가 많다.
```python
import copy, re
from docx.oxml.ns import qn
W = lambda t: qn("w:" + t)

def ptext(p):                       # 문단 글자 전체
    return "".join(t.text or "" for t in p.iter(W("t")))

def preplace(p, old, new, count=1):
    """문단 p 안의 old를 new로. 첫 조각의 서식을 쓰고 나머지 조각에서는 지운다."""
    done = 0
    while done < count:
        ts = list(p.iter(W("t"))); full = "".join(t.text or "" for t in ts)
        i = full.find(old)
        if i < 0: break
        j = i + len(old); pos = 0; first = True
        for t in ts:
            s = t.text or ""; a, b = pos, pos + len(s); pos = b
            if b <= i or a >= j: continue
            lo, hi = max(i, a) - a, min(j, b) - a
            t.text = (s[:lo] + new + s[hi:]) if first else (s[:lo] + s[hi:]); first = False
            t.set("{http://www.w3.org/XML/1998/namespace}space", "preserve")
        done += 1
        if old in new: break        # 새 글이 옛 글을 포함하면 무한 반복 방지
    return done
```
사용 예
```python
import docx
d = docx.Document("보관본.docx")
for p in d.element.body.iter(W("p")):
    if "회수 여부 ##" in ptext(p):
        preplace(p, "(신청서 제출일 현재 회수 여부 ##)", "")
d.save("다음판.docx")
```

## 2. 대기 표식만 노란색으로 (##, ##.##, [확인 필요: …], [작성 대기: …])
```python
from docx.oxml import OxmlElement
PEND = re.compile(r"\[확인 필요[^\]]*\]|\[작성 대기[^\]]*\]|#+(?:\.#+)*")

def set_hl(r, color=None):          # color=None이면 형광펜 제거
    rpr = r.find(W("rPr"))
    if rpr is None: rpr = OxmlElement("w:rPr"); r.insert(0, rpr)
    h = rpr.find(W("highlight"))
    if color is None:
        if h is not None: rpr.remove(h)
        return
    if h is None: h = OxmlElement("w:highlight"); rpr.append(h)
    h.set(W("val"), color)

def mark_pending(p):
    """런을 쪼개 표식 부분만 노란색. 나머지 서식은 원래 런 것을 복사."""
    for r in list(p.iter(W("r"))):
        t = r.find(W("t"))
        if t is None or not PEND.search(t.text or ""): continue
        rpr = r.find(W("rPr")); parent = r.getparent(); idx = parent.index(r)
        s = t.text; last = 0; pieces = []
        for m in PEND.finditer(s):
            if m.start() > last: pieces.append((s[last:m.start()], False))
            pieces.append((m.group(0), True)); last = m.end()
        if last < len(s): pieces.append((s[last:], False))
        for n, (txt, hl) in enumerate(pieces):
            nr = OxmlElement("w:r")
            if rpr is not None: nr.append(copy.deepcopy(rpr))
            nt = OxmlElement("w:t"); nt.text = txt
            nt.set("{http://www.w3.org/XML/1998/namespace}space", "preserve"); nr.append(nt)
            set_hl(nr, "yellow" if hl else None)
            parent.insert(idx + n, nr)
        parent.remove(r)
```
회사가 칠한 형광펜을 모두 빼고 표식만 노랗게 하려면: 해당 문단의 모든 런에 `set_hl(r, None)` → `mark_pending(p)`.

## 3. 칸(셀) 안을 의미 단위 여러 줄로 다시 쓰기
**주의:** 첫 문단을 통째로 복사해 둘째 줄을 만들면 첫 줄 글자까지 복사된다. 빈 틀에서 만든다.
```python
def rewrite_cell(tc, lines):
    ps = tc.findall(W("p")); first = ps[0]
    ppr = first.find(W("pPr"))
    r0 = first.find(".//" + W("r")); rpr = r0.find(W("rPr")) if r0 is not None else None
    ppr, rpr = (copy.deepcopy(x) if x is not None else None for x in (ppr, rpr))   # 지우기 전에 서식 복사
    for p in ps: tc.remove(p)
    for line in lines:
        p = OxmlElement("w:p")
        if ppr is not None: p.append(copy.deepcopy(ppr))
        r = OxmlElement("w:r")
        if rpr is not None: r.append(copy.deepcopy(rpr))
        t = OxmlElement("w:t"); t.text = line
        t.set("{http://www.w3.org/XML/1998/namespace}space", "preserve"); r.append(t)
        p.append(r); tc.append(p)

# 예: rewrite_cell(tc, ["대표이사", "(등기)"])
```

## 4. Word 메모(코멘트) 지우기 — 부속 파일까지
```python
from lxml import etree
W14 = "http://schemas.microsoft.com/office/word/2010/wordml"
W15 = "http://schemas.microsoft.com/office/word/2012/wordml"
CID = "http://schemas.microsoft.com/office/word/2016/wordml/cid"
CEX = "http://schemas.microsoft.com/office/word/2018/wordml/cex"

def part(d, name):
    for p in d.part.package.iter_parts():
        if str(p.partname) == name: return p

def delete_comments(d, ids):
    """ids: 지울 메모 번호(문자열) 집합. 본문 표시, 메모 본체, 부속 3종을 함께 지운다."""
    body = d.element.body; cp = d.part._comments_part.element
    for x in list(body.iter(W("commentRangeStart"), W("commentRangeEnd"))):
        if x.get(W("id")) in ids: x.getparent().remove(x)
    for x in list(body.iter(W("commentReference"))):
        if x.get(W("id")) in ids: r = x.getparent(); r.getparent().remove(r)
    paras = set()
    for c in list(cp.findall(W("comment"))):
        if c.get(W("id")) in ids:
            ps = c.findall(W("p"))
            if ps: paras.add(ps[-1].get("{%s}paraId" % W14))
            cp.remove(c)
    durable = set()
    for name, attr, key in (("/word/commentsIds.xml", "{%s}paraId" % CID, "paras"),
                            ("/word/commentsExtended.xml", "{%s}paraId" % W15, "paras"),
                            ("/word/commentsExtensible.xml", "{%s}durableId" % CEX, "durable")):
        p = part(d, name)
        if p is None: continue
        root = etree.fromstring(p.blob)
        for x in list(root):
            if (key == "paras" and x.get(attr) in paras) or (key == "durable" and x.get(attr) in durable):
                if name.endswith("Ids.xml"): durable.add(x.get("{%s}durableId" % CID))
                root.remove(x)
        p._blob = etree.tostring(root, xml_declaration=True, encoding="UTF-8", standalone=True)

def orphan_comments(d):
    """본문에 연결이 없는 메모 번호(문단을 지우거나 합친 뒤 생김)."""
    ref = {x.get(W("id")) for x in d.element.body.iter(W("commentRangeStart"), W("commentReference"))}
    return {c.get(W("id")) for c in d.part._comments_part.element.findall(W("comment"))} - ref
```
메모 작성자, 내용 보기: `for c in d.comments: print(c.comment_id, c.author, c.text)` (python-docx 1.2 이상).

## 5. 장 구분과 판 검증
```python
CH = re.compile(r"^([ⅠⅡⅢⅣⅤⅥⅦⅧⅨ])\.\s")
def chapters(d):
    """장별 본문 요소. 제목 스타일 id가 '1' 또는 'Heading1'일 수 있다."""
    out, cur = {}, "앞"
    for el in d.element.body.iterchildren():
        if el.tag == W("p"):
            s = ptext(el).strip(); ps = el.find(W("pPr") + "/" + W("pStyle"))
            m = CH.match(s)
            if m and ps is not None and ps.get(W("val")) in ("1", "Heading1"): cur = m.group(1)
        if el.tag != W("sectPr"): out.setdefault(cur, []).append(el)
    missing = [c for c in "ⅠⅡⅢⅤⅥⅦ" if c not in out]
    assert not missing, f"장 제목을 못 찾음: {missing}"      # 못 찾으면 멈춘다
    return out

def cells_and_paras(els):
    out = []
    for e in els:
        if e.tag == W("p"): out.append(("문단", ptext(e)))
        elif e.tag == W("tbl"):
            for tc in e.iter(W("tc")): out.append(("칸", " / ".join(ptext(p) for p in tc.findall(W("p")))))
    return out

def compare(before_docx, after_docx, untouched=("Ⅳ",)):
    import difflib
    a, b = chapters(docx.Document(before_docx)), chapters(docx.Document(after_docx))
    for ch in a:
        if ch in untouched:      # 다른 팀 장: XML까지 같아야 한다
            same = len(a[ch]) == len(b[ch]) and all(etree.tostring(x) == etree.tostring(y) for x, y in zip(a[ch], b[ch]))
            print(ch, "XML 동일" if same else "!! 바뀜"); continue
        x, y = cells_and_paras(a[ch]), cells_and_paras(b[ch])
        for op, i1, i2, j1, j2 in difflib.SequenceMatcher(None, x, y, autojunk=False).get_opcodes():
            if op == "equal": continue
            for u in x[i1:i2]: print(ch, "전", u[1][:150])
            for u in y[j1:j2]: print(ch, "후", u[1][:150])
```
판마다 확인: `compare(...)` 출력이 의도한 것뿐인지, `orphan_comments(d)`가 비었는지, 메모 수, 그림 수(`len(d.inline_shapes)`), 「@」 0건.

## 6. 글자 폭 재기와 열 폭 정하기
```python
from PIL import ImageFont
# 맑은 고딕: 맥은 /Applications/Microsoft Word.app/Contents/Resources/DFonts/malgun.ttf, 윈도는 C:/Windows/Fonts/malgun.ttf
FR = ImageFont.truetype("malgun.ttf 경로", 1000); FB = ImageFont.truetype("malgunbd.ttf 경로", 1000)

def width_dxa(text, pt=9, bold=False):    # Word 표 폭 단위(1/20pt)
    return (FB if bold else FR).getlength(text) / 1000 * pt * 20 * 1.03 + 20   # 3% + 20 여유

# 칸 안쪽 폭 = 열 폭 - 좌우 셀 여백(예: 108+108)
# 열 최소폭 = 그 열의 「쪼갤 수 없는 단위」(숫자, 날짜, 괄호 묶음, 의미 단위) 중 가장 긴 것 + 여백
# 남는 폭은 줄바꿈이 생기는 긴 글 열(약력 등)에 몰아준다. 표 전체 폭은 유지
```

## 7. Word가 「읽을 수 없는 내용」 오류를 내지 않게 (XML 자식 순서)
새 서식 요소를 덧붙이면 순서가 어긋날 수 있다. 고친 표, 문단에만 적용한다.
```python
ORD = {
 "pPr": "pStyle keepNext keepLines pageBreakBefore framePr widowControl numPr suppressLineNumbers pBdr shd tabs suppressAutoHyphens kinsoku wordWrap overflowPunct topLinePunct autoSpaceDE autoSpaceDN bidi adjustRightInd snapToGrid spacing ind contextualSpacing mirrorIndents suppressOverlap jc textDirection textAlignment textboxTightWrap outlineLvl divId cnfStyle rPr sectPr pPrChange",
 "rPr": "rStyle rFonts b bCs i iCs caps smallCaps strike dstrike outline shadow emboss imprint noProof snapToGrid vanish webHidden color spacing w kern position sz szCs highlight u effect bdr shd fitText vertAlign rtl cs em lang eastAsianLayout specVanish oMath",
 "tcPr": "cnfStyle tcW gridSpan hMerge vMerge tcBorders shd noWrap tcMar textDirection tcFitText vAlign hideMark headers cellIns cellDel cellMerge tcPrChange",
}
def fix_order(root):
    for tag, order in ORD.items():
        rank = {qn("w:" + t): i for i, t in enumerate(order.split())}
        for el in root.iter(qn("w:" + tag)):
            kids = list(el); srt = sorted(kids, key=lambda k: rank.get(k.tag, 999))
            if srt != kids:
                for k in kids: el.remove(k)
                for k in srt: el.append(k)
```

## 9. 엑셀 다루기
- 사용자가 고친 엑셀에는 **새 줄만 덧붙인다**(openpyxl로 마지막 행 다음에, 위 행 서식 복사).
- 그림, 도형이 든 엑셀(`xl/drawings/` 가 있는지 zip으로 확인)은 openpyxl로 저장하지 않는다(그림이 사라질 수 있음). 별도 파일을 만든다.
```python
import zipfile
has_drawings = any(n.startswith("xl/drawings/") for n in zipfile.ZipFile("리스트.xlsx").namelist())
```
- 엑셀 안 숨김 시트의 그림(계약서 스캔 등) 꺼내기: 시트 → `xl/worksheets/_rels/sheetN.xml.rels`의 drawing → `xl/drawings/_rels/drawingN.xml.rels`의 media 목록을 따라가 `xl/media/`에서 꺼낸다.
